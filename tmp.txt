devices=(--device /dev/davinci0 --device /dev/davinci_manager)
for node in /dev/hisi_hdc /dev/ummu /dev/uburma; do
  if [ -e "$node" ]; then
    devices+=(--device "$node")
  fi
done

docker run --rm --pull=never --shm-size=1g \
  "${devices[@]}" \
  -v /usr/local/Ascend/driver:/usr/local/Ascend/driver:ro \
  -v /usr/local/dcmi:/usr/local/dcmi:ro \
  -v "$(command -v npu-smi):/usr/local/bin/npu-smi:ro" \
  -v /etc/ascend_install.info:/etc/ascend_install.info:ro \
  -v "$PWD:/fgw-mhc" -w /fgw-mhc --entrypoint python \
  quay.io/ascend/vllm-ascend:v0.30.0-a5 \
  bench_vllm_ascend.py --backend vllm --device npu:0 --output vllm-pair.csv &&
python bench_vllm_ascend.py --merge tilelang-pair.csv vllm-pair.csv \
  --output vllm-ascend-results.csv &&
cat vllm-ascend-results.csv
