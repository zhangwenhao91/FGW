devices=(--device /dev/davinci_manager)
for id in {0..7}; do
  node="/dev/davinci$id"
  if [ -c "$node" ]; then devices+=(--device "$node"); fi
done

  if [ -c "$node" ]; then devices+=(--device "$node"); fi
done
