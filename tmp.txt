if [ ! -f ../vllm/pyproject.toml ]; then
  git clone --depth 1 --branch v0.32.0 \
    https://github.com/vllm-project/vllm.git ../vllm
fi

constraints=$(mktemp)
python - <<'PY' > "$constraints"
from importlib.metadata import version

for name in ("torch", "torch-npu"):
    print(f"{name}=={version(name)}")
PY

python - <<'PY' | python -m pip install -c "$constraints" -r /dev/stdin
from pathlib import Path
import re
import tomllib

for path in ("../vllm/pyproject.toml", "../vllm-ascend/pyproject.toml"):
    for req in tomllib.loads(Path(path).read_text())["build-system"]["requires"]:
        name = re.split(r"[\s<>=!~;\[]", req, maxsplit=1)[0].replace("_", "-")
        if name not in {"torch", "torch-npu", "torchvision", "torchaudio"}:
            print(req)
PY
