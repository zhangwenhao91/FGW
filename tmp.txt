ASCEND_RT_VISIBLE_DEVICES=4 python bench_vllm_ascend.py \
  --backend tilelang --projection cube_full --device npu:0 \
  --output tilelang-adaptive-pair-card4.csv

devices=(--device /dev/davinci4 --device /dev/davinci_manager)
for node in /dev/hisi_hdc /dev/ummu /dev/uburma; do
  if [ -e "$node" ]; then devices+=(--device "$node"); fi
done

docker run --rm --pull=never --shm-size=1g "${devices[@]}" \
  -e ASCEND_RT_VISIBLE_DEVICES=4 \
  -v /usr/local/Ascend/driver:/usr/local/Ascend/driver:ro \
  -v /usr/local/dcmi:/usr/local/dcmi:ro \
  -v "$(command -v npu-smi):/usr/local/bin/npu-smi:ro" \
  -v /etc/ascend_install.info:/etc/ascend_install.info:ro \
  -v "$PWD:/fgw-mhc" -w /fgw-mhc --entrypoint python \
  quay.io/ascend/vllm-ascend:v0.30.0-a5 \
  bench_vllm_ascend.py --backend vllm --device npu:0 \
  --output vllm-pair-card4.csv &&
python bench_vllm_ascend.py \
  --merge tilelang-adaptive-pair-card4.csv vllm-pair-card4.csv \
  --output vllm-adaptive-results-card4.csv &&
cat vllm-adaptive-results-card4.csv
